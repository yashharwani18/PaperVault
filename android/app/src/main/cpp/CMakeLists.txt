cmake_minimum_required(VERSION 3.22)
project(local_library LANGUAGES C CXX)
set(CMAKE_C_FLAGS_DEBUG "-O3 -g0 -DNDEBUG")
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g0 -DNDEBUG")
set(CMAKE_C_FLAGS_RELWITHDEBINFO "-O3 -g0 -DNDEBUG")
set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "-O3 -g0 -DNDEBUG")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "x86_64")
    # The test emulator advertises AVX2.
    set(GGML_SSE42 ON CACHE BOOL "" FORCE)
    set(GGML_AVX ON CACHE BOOL "" FORCE)
    set(GGML_AVX2 ON CACHE BOOL "" FORCE)
    set(GGML_FMA ON CACHE BOOL "" FORCE)
    set(GGML_F16C ON CACHE BOOL "" FORCE)
endif()
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    # Runtime feature checks select dot-product / I8MM when supported.
    # The ARMv8 baseline remains available for older phones.
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
else()
    set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
endif()
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
option(KHOJ_KLEIDIAI "Enable pinned KleidiAI ARM kernels with runtime feature checks" ON)
if(ANDROID_ABI STREQUAL "arm64-v8a" AND KHOJ_KLEIDIAI)
    set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
else()
    set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
endif()
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_library(library_models SHARED models.cpp)
option(KHOJ_EXPERIMENTAL_NATIVE_BATCHES "Benchmark unvalidated multi-sequence embeddings" OFF)
if(KHOJ_EXPERIMENTAL_NATIVE_BATCHES)
    target_compile_definitions(library_models PRIVATE KHOJ_EXPERIMENTAL_NATIVE_BATCHES=1)
endif()
target_include_directories(library_models PRIVATE "${LLAMA_SOURCE}/include")
target_link_libraries(library_models PRIVATE llama android log)
target_link_options(library_models PRIVATE "-Wl,-z,max-page-size=16384")
